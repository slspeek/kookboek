\begin{recipe}
[ %
    preparationtime = {\unit[25]{min}},
    bakingtime = {\unit[45]{min} + \unit[20]{min} oventijd},
    bakingtemperature = {\protect\bakingtemperature{
	    fanoven=\unit[180]{\textcelcius}, topbottomheat=\unit[200]{\textcelcius}}},
    portion = {\portion{4}},
    % source = {\href{https://www.leukerecepten.nl/recepten/gezonde-pastasalade/}{Leuke recepten}}
]
{Aubergine, pompoen, halloumi \& bulgur}

    \ingredients
    {%
	  \unit[700]{g} & pompoen\\
      1 grote & aubergine\\
      \unit[250]{g} & halloumi\\
      \unit[4]{el} & olijfolie\\
      1 bosje & munt \\
      \unit[200]{g} & bulgur \\
      \unit[400]{ml} & groentebouillon \\ 
	    & zout \\
        & peper \\
    }

    \preparation
    {%
	    \step Verwarm de oven voor.
	    \step Snijd de pompoen, aubergine en halloumi in blokjes van ongeveer 1 cm.
        \step Snijd de munt fijn.
        \step Doe de pompoen, aubergine, halloumi met de helft van de munt
        in een ovenschaal, met de olijfolie en bestrooi
        met flink wat peper en zout.
        \step Rooster de pompoen, aubergine en halloumi in de oven voor ongeveer 45 minuten.
        \step Warm de groentebouillon op in een pan.
        \step Meng de bulgur door de groenten heen en giet de warme bouillon eroverheen.
        \step Zet de schaal afgedekt met aluminiumfolie in de oven voor ongeveer 20 minuten.
        \step Haal de schaal uit de oven en laat haar 5 minuten afgedekt staan.
        \step Serveer de bulgur met de geroosterde groenten en halloumi, bestrooid met de rest van de munt.
    }
    
\end{recipe}
