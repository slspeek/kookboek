\begin{recipe}
[ %
    preparationtime = {\unit[45]{min}},
    portion = {\portion{4}},
]
{Pasta met linzen}

    \ingredients
    {%
	\unit[1]{bakje} & gemengde paddenstoelen\\
    \unit[1]{blikje} & linzen\\
    \unit[200]{g} & cr\`{e}me fra\^{i}che\\ 
    \unit[1]{zakje} & rucola\\
    \unit[2]{zakjes} & gevulde pasta met spinazie\\
    1 & ui\\
    \unit[4]{teentjes} & knoflook\\
    \unit[120]{g} & pecannoten\\
    \unit[2]{el} & olijfolie \\
    & zout \\
    & peper \\
    }

    \preparation
    {%
	    \step Snipper de ui en pel de knoflook.
        \step Snijd de paddenstoelen grof en hak de pecannoten.
        \step Smoor de ui, de gepeste knoflook en paddenstoelen 15 minuten in de olijfolie. Voeg peper en zout toe.
        \step Kook de pasta in water met zout.
        \step Op het laatste het noten gehakt, cr\`{e}me fra\^{i}che en linzen toevoegen en warm maken op hoger vuur.
        \step Serveer de pasta op de borden met daarop de rucola, en doe de saus daaroverheen.
    }
    
\end{recipe}
