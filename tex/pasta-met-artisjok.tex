\begin{recipe}
[ %
    preparationtime = {\unit[30]{min}},
    portion = {\portion{2}},
]
{Pasta met artisjok en citroen}

    \ingredients
    {%
        \unit[200]{g} & tagliatelle \\
        2 & sjalotten \\
        \unit[200]{g} & artisjokharten \\
        \unit[100]{ml} & kookroom \\
        \unit[25]{g} & Parmezaanse kaas \\
        handje & peterselie \\
        halve & citroen \\
        \unit[1]{el} & olijfolie \\
	    & zout \\
        & peper \\
    }

    \preparation
    {%
	    \step Breng water met zout aan de kook. En kook de pasta gaar.
        \step Hak de peterselie fijn.
	    \step Snijd de sjalotten en knoflook fijn. Snijd de artisjokharten in vieren.
        \step Verhit de olijfolie in een hapjespan op middelhoog vuur.
        \step Fruit de sjalot en knoflook tot ze glazig zijn.
        \step Bak de artisjok 5 minuten mee.
        \step Rasp de citroenschil en pers het sap uit. Voeg dat toe aan de pan.
        \step Voeg de kookroom toe en laat de saus 5 minuten zachtjes koken.
        \step Rasp de Parmezaanse kaas.
        \step Giet de tagliatelle af. En voeg die toe aan de saus, roer de Parmezaanse kaas erdoor.
        \step Serveer met de peterselie.
    }
    
\end{recipe}
