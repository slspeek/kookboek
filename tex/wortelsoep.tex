\begin{recipe}
[ %
    preparationtime = {\unit[45]{min}},
    bakingtime = {\unit[20]{min} koken},
    portion = {\portion{3}},
]
{Wortelsoep}

    \ingredients
    {%
        \unit[4]{tenen} & knoflook \\
        3 & uien \\
        \unit[1]{pond} & wortelen \\
        \unit[1]{pond} & zoete tomaten \\
        \unit[150]{g} & rode linzen \\
        \unit[1]{pakje} & gerookte tofoe/temp\'{e} \\ 
        2 & bouillonblokjes \\
        \unit[1]{l} & water \\
        \unit[2]{el} & olijfolie \\
        & volkoren stokbrood \\
        & peper \\
    }

    \preparation
    {%
	    \step Snijd de knoflook fijn en snipper de ui.
        \step Fruit de knoflook en ui in een soeppan met olijfolie.
        \step Snijd de wortelen klein. En snijd de tomaten in blokjes.
        \step Voeg de wortelen, tomaten, linzen, bouillonblokjes en het water toe aan de pan.
        Breng het geheel aan de kook en laat het 20 minuten koken.        
        \step Snijd de tofoe/temp\'{e} in blokjes en bak deze in een koekenpan.
	    \step Pureer de soep met een staafmixer en breng op smaak met peper.
        \step Serveer de soep met de gebakken tofoe/temp\'{e} en stokbrood.
    }
    
\end{recipe}
